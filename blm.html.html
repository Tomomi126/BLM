        function updateDate(lang = 'vi') {

            const dateElement = document.getElementById("current-date");

            const today = new Date();

            const optionsVI = { weekday: 'long', day: '2-digit', month: 'long', year: 'numeric' };

            const optionsEN = { weekday: 'long', day: '2-digit', month: 'long', year: 'numeric' };

            if (lang === 'en') {

                const days = {

                    "Chủ Nhật": "Sunday", "Thứ Hai": "Monday", "Thứ Ba": "Tuesday", "Thứ Tư": "Wednesday",

                    "Thứ Năm": "Thursday", "Thứ Sáu": "Friday", "Thứ Bảy": "Saturday"

                };

                const months = {

                    "Tháng Một": "January", "Tháng Hai": "February", "Tháng Ba": "March", "Tháng Tư": "April",

                    "Tháng Năm": "May", "Tháng Sáu": "June", "Tháng Bảy": "July", "Tháng Tám": "August",

                    "Tháng Chín": "September", "Tháng Mười": "October", "Tháng Mười Một": "November", "Tháng Mười Hai": "December"

                };

                let dateStr = today.toLocaleDateString('vi-VN', optionsVI);

                Object.keys(days).forEach(vn => {

                    dateStr = dateStr.replace(vn, days[vn]);

                });

                Object.keys(months).forEach(vn => {

                    dateStr = dateStr.replace(vn, months[vn]);

                });

                dateElement.textContent = dateStr;

            } else {

                dateElement.textContent = today.toLocaleDateString('vi-VN', optionsVI);

            }

        }

        updateDate(); // Gọi hàm cập nhật khi trang tải

        const translations = {

            vi: {

                heading: "Chào mừng bạn đến với trang web của chúng tôi!",

                description: "Đây là một trang web thử nghiệm giúp bạn đổi ngôn ngữ giữa Tiếng Việt và Tiếng Anh.",

                moreInfo: "Nhấn vào lá cờ để thay đổi ngôn ngữ hiển thị.",

                langVi: "Tiếng Việt",

                langEn: "English"

            },

            en: {

                heading: "Welcome to our website!",

                description: "This is a demo website that allows you to switch between Vietnamese and English.",

                moreInfo: "Click on the flag to change the display language.",

                langVi: "Vietnamese",

                langEn: "English"

            }

        };

        function changeLanguage(lang) {

            document.getElementById("heading").textContent = translations[lang].heading;

            document.getElementById("description").textContent = translations[lang].description;

            document.getElementById("more-info").textContent = translations[lang].moreInfo;

            document.getElementById("text-vi").textContent = translations[lang].langVi;

            document.getElementById("text-en").textContent = translations[lang].langEn;

            updateDate(lang);

        }

    </script>

    function updatePageViews() {

        let today = new Date().toDateString(); // Ngày hiện tại

        let storedDate = localStorage.getItem("lastVisitDate"); // Lần truy cập cuối

        let viewsToday = parseInt(localStorage.getItem("viewsToday")) || 0;

        let totalVisits = parseInt(localStorage.getItem("totalVisits")) || 0;

        let pageViews = parseInt(localStorage.getItem("pageViews")) || 0;

        if (storedDate !== today) {

            // Nếu ngày đã thay đổi, reset "Lượt xem hôm nay"

            viewsToday = 0;

            localStorage.setItem("lastVisitDate", today);

        }

        viewsToday++; // Tăng lượt xem hôm nay

        totalVisits++; // Tăng tổng truy cập

        pageViews++; // Tăng tổng lượt xem trang

        // Lưu vào localStorage

        localStorage.setItem("viewsToday", viewsToday);

        localStorage.setItem("totalVisits", totalVisits);

        localStorage.setItem("pageViews", pageViews);

        // Cập nhật UI

        document.getElementById("views-today").innerText = viewsToday;

        document.getElementById("total-visits").innerText = totalVisits;

        document.getElementById("page-views").innerText = pageViews;

    }

    // Gọi hàm cập nhật khi trang được tải

    updatePageViews();

    // Hiển thị nút quay lại đầu trang khi cuộn xuống

    window.onscroll = function() {

        var btn = document.querySelector(".back-to-top");

        if (document.documentElement.scrollTop > 200) {

            btn.style.display = "block";

        } else {

            btn.style.display = "none";

        }

    };

    // Hàm cuộn lên đầu trang

    function scrollToTop() {

        window.scrollTo({top: 0, behavior: 'smooth'});

    }

</script>
